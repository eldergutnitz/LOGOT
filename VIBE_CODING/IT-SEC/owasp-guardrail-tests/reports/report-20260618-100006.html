ADDRESS_RE = re.compile(r"\b\d{1,6}\s+[A-Za-z0-9.\- ]+\s+(?:St|Street|Ave|Avenue|Rd|Road|Blvd|Boulevard|Ln|Lane|Dr|Drive)\b",
                        re.IGNORECASE)

@dataclass(frozen=True)
class ScrubResult:
    redacted_text: str
    pii_counts: Dict[str, int]

def _luhn_ok(number: str) -> bool:
    digits = [int(c) for c in number if c.isdigit()]
    if len(digits) < 13 or len(digits) > 19:
        return False
    checksum = 0
    parity = len(digits) % 2
    for i, d in enumerate(digits):
        if i % 2 == parity:
            d *= 2
            if d > 9:
                d -= 9
        checksum += d
    return checksum % 10 == 0

        raise TypeError("text must be a string")

        raise ValueError("text too large")

    def sub_and_count(pattern: re.Pattern, label: str, s: str, repl: str) -> str:
        matches = list(pattern.finditer(s))
        if matches:
            counts[label] = counts.get(label, 0) + len(matches)
            s = pattern.sub(repl, s)
        return s

    redacted = sub_and_count(PHONE_RE, "phone", redacted, "[REDACTED:PHONE]")
    redacted = sub_and_count(SSN_RE, "ssn", redacted, "[REDACTED:SSN]")
    redacted = sub_and_count(IPV4_RE, "ip", redacted, "[REDACTED:IP]")
    redacted = sub_and_count(ADDRESS_RE, "address", redacted, "[REDACTED:ADDRESS]")

    # Credit cards: validate candidates with Luhn before redacting
    cc_hits = []
    for m in CC_CANDIDATE_RE.finditer(redacted):
        candidate = m.group(0)
        if _luhn_ok(candidate):
            cc_hits.append((m.start(), m.end()))

        out = redacted
        for start, end in reversed(cc_hits):
            out = out[:start] + "[REDACTED:CREDIT_CARD]" + out[end:]
        redacted = out

/**
 * Deterministic keying; avoid storing raw PII in Redis keys (LLM02).
 * Hash userId/ip to reduce accidental disclosure in logs/metrics.
 */
export function hashId(value) {
  return crypto.createHash("sha256").update(String(value)).digest("hex").slice(0, 32);
}

/**
 * Fixed-window counter with TTL.
 * Good enough for gateways; for stricter fairness use sliding window or token bucket.
 */
export async function incrWithTtl(redis, key, windowSeconds) {
  const multi = redis.multi();
  multi.incr(key);
  multi.ttl(key);
  const [[, count], [, ttl]] = await multi.exec();

  if (ttl === -1) {
    await redis.expire(key, windowSeconds);
  }
  return count;
}

/**
 * Enforce a max count per window.
 */
export async function enforceRateLimit({ redis, key, limit, windowSeconds }) {
  const count = await incrWithTtl(redis, key, windowSeconds);
  if (count > limit) {
    const retryAfter = await redis.ttl(key);
    const err = new Error("Rate limit exceeded");
    err.status = 429;
    err.headers = { "Retry-After": String(Math.max(1, retryAfter)) };
    throw err;

}
