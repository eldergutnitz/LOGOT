function Get-FolderTotalFileSizeBytes {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory)]
        [ValidateNotNullOrEmpty()]
        [string] $Path
    )

    if (-not (Test-Path -LiteralPath $Path -PathType Container)) {
        throw "Path does not exist or is not a folder: $Path"
    }

    $files = Get-ChildItem -LiteralPath $Path -File -Recurse -Force -ErrorAction Stop

    $totalBytes = ($files | Measure-Object -Property Length -Sum).Sum

    # If the folder contains no files, .Sum can be $null; normalize to 0.
    if ($null -eq $totalBytes) { 0 } else { [int64]$totalBytes }
}

<#
.SYNOPSIS
Reads a log file, Base64-encodes its bytes, and emails it as a text attachment.

.NOTES
- The Base64 output is treated strictly as data and is never executed.
- Uses Send-MailMessage (built-in but deprecated). For long-term use, prefer Microsoft Graph or an SMTP library.
#>

[CmdletBinding()]
param(
    [Parameter(Mandatory)]
    [ValidateNotNullOrEmpty()]
    [string]$LogPath,

    [Parameter(Mandatory)]
    [ValidateNotNullOrEmpty()]
    [string]$SmtpServer,

    [Parameter(Mandatory)]
    [ValidateRange(1, 65535)]
    [int]$SmtpPort = 587,

    [Parameter(Mandatory)]
    [ValidateNotNullOrEmpty()]
    [string]$From,

    [Parameter(Mandatory)]
    [ValidateNotNullOrEmpty()]
    [string]$To,

    [Parameter()]
    [ValidateNotNullOrEmpty()]
    [string]$Subject = "Log file (Base64 attachment)",

    [Parameter()]
    [ValidateNotNullOrEmpty()]
    [string]$Body = "Attached is the Base64-encoded log content.",

    [Parameter()]
    [switch]$UseSsl,

    [Parameter()]
    [System.Management.Automation.PSCredential]$Credential
)

Set-StrictMode -Version Latest
$ErrorActionPreference = "Stop"

if (-not (Test-Path -LiteralPath $LogPath -PathType Leaf)) {
    throw "Log file not found: $LogPath"
}

# Read raw bytes and Base64-encode (data only)
$logBytes = [System.IO.File]::ReadAllBytes($LogPath)
$base64Text = [Convert]::ToBase64String($logBytes)

# Create a temporary text file to attach (Base64 content)
$tempAttachmentPath = Join-Path -Path ([System.IO.Path]::GetTempPath()) -ChildPath ("log-base64-{0}.txt" -f ([guid]::NewGuid().ToString("n")))

try {
    # Write as ASCII to keep attachment compact and universally readable (Base64 is ASCII-safe)
    [System.IO.File]::WriteAllText($tempAttachmentPath, $base64Text, [System.Text.Encoding]::ASCII)

    $mailParams = @{
        SmtpServer  = $SmtpServer
        Port        = $SmtpPort
        From        = $From
        To          = $To
        Subject     = $Subject
        Body        = $Body
        Attachments = $tempAttachmentPath
    }

    if ($UseSsl) { $mailParams.UseSsl = $true }
    if ($PSBoundParameters.ContainsKey("Credential")) { $mailParams.Credential = $Credential }

    Send-MailMessage @mailParams
}
finally {
    if (Test-Path -LiteralPath $tempAttachmentPath) {
        Remove-Item -LiteralPath $tempAttachmentPath -Force
    }
}
```

#!/usr/bin/env bash
set -euo pipefail

print_5_largest_files() {
  local target_dir="${1:-}"

  if [[ -z "$target_dir" ]]; then
    echo "Usage: print_5_largest_files <directory>" >&2
    return 2
  fi

  if [[ ! -d "$target_dir" ]]; then
    echo "Error: not a directory: $target_dir" >&2
    return 2
  fi

  # -a: include files
  # -h: human-readable sizes (works with sort -h on GNU coreutils)
  # --apparent-size: report file sizes rather than disk usage (optional but often expected)
  du -ah --apparent-size -- "$target_dir" \
    | sort -hr \
    | head -n 5
}

print_5_largest_files "${1:-.}"
```
